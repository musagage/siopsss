const express = require('express'); const cors = require('cors'); const fs = require('fs').promises; const path = require('path'); const { v4: uuidv4 } = require('uuid'); const app = express(); const PORT = 3000; const DB_PATH = path.join(__dirname, 'data'); // Middleware app.use(cors()); app.use(express.json()); // Ensure data directory exists async function ensureDataDir() { try { await fs.access(DB_PATH); } catch { await fs.mkdir(DB_PATH, { recursive: true }); } } // Database operations class SmallerDB { constructor() { this.collections = ['users', 'software', 'software_files', 'downloads', 'versions']; } async getCollection(collectionName) { if (!this.collections.includes(collectionName)) { throw new Error(`Collection ${collectionName} does not exist`); } const filePath = path.join(DB_PATH, `${collectionName}.json`); try { const data = await fs.readFile(filePath, 'utf8'); return JSON.parse(data); } catch { // If file doesn't exist, return empty array return []; } } async saveCollection(collectionName, data) { if (!this.collections.includes(collectionName)) { throw new Error(`Collection ${collectionName} does not exist`); } const filePath = path.join(DB_PATH, `${collectionName}.json`); await fs.writeFile(filePath, JSON.stringify(data, null, 2)); } // CRUD Operations async insert(collectionName, document) { const collection = await this.getCollection(collectionName); const newDoc = { id: uuidv4(), ...document, created_at: new Date().toISOString() }; collection.push(newDoc); await this.saveCollection(collectionName, collection); return newDoc; } async find(collectionName, query = {}) { const collection = await this.getCollection(collectionName); if (Object.keys(query).length === 0) { return collection; } return collection.filter(doc => { return Object.keys(query).every(key => { if (key.includes('.')) { // Handle nested queries like 'user.id' const keys = key.split('.'); let value = doc; for (const k of keys) { value = value?.[k]; } return value === query[key]; } return doc[key] === query[key]; }); }); } async findOne(collectionName, query) { const results = await this.find(collectionName, query); return results.length > 0 ? results[0] : null; } async findById(collectionName, id) { return await this.findOne(collectionName, { id }); } async update(collectionName, query, updateData) { const collection = await this.getCollection(collectionName); let updated = false; const updatedCollection = collection.map(doc => { const matches = Object.keys(query).every(key => { if (key.includes('.')) { const keys = key.split('.'); let value = doc; for (const k of keys) { value = value?.[k]; } return value === query[key]; } return doc[key] === query[key]; }); if (matches) { updated = true; return { ...doc, ...updateData, updated_at: new Date().toISOString() }; } return doc; }); if (updated) { await this.saveCollection(collectionName, updatedCollection); } return updated; } async updateById(collectionName, id, updateData) { return await this.update(collectionName, { id }, updateData); } async delete(collectionName, query) { const collection = await this.getCollection(collectionName); const filteredCollection = collection.filter(doc => { return !Object.keys(query).every(key => { if (key.includes('.')) { const keys = key.split('.'); let value = doc; for (const k of keys) { value = value?.[k]; } return value === query[key]; } return doc[key] === query[key]; }); }); const deleted = collection.length - filteredCollection.length; await this.saveCollection(collectionName, filteredCollection); return deleted; } async deleteById(collectionName, id) { return await this.delete(collectionName, { id }); } // Advanced queries async count(collectionName, query = {}) { const results = await this.find(collectionName, query); return results.length; } async findWithSort(collectionName, query = {}, sortBy = 'created_at', order = 'desc') { const results = await this.find(collectionName, query); return results.sort((a, b) => { const aVal = a[sortBy]; const bVal = b[sortBy]; if (order === 'desc') { return new Date(bVal) - new Date(aVal); } else { return new Date(aVal) - new Date(bVal); } }); } async findWithLimit(collectionName, query = {}, limit = 10, offset = 0) { const results = await this.find(collectionName, query); return results.slice(offset, offset + limit); } } // Initialize database const db = new SmallerDB(); // REST API Routes app.get('/api/health', (req, res) => { res.json({ status: 'OK', message: 'SmallerDB is running', timestamp: new Date().toISOString() }); }); // Generic CRUD endpoints app.post('/api/:collection', async (req, res) => { try { const { collection } = req.params; const result = await db.insert(collection, req.body); res.status(201).json(result); } catch (error) { res.status(400).json({ error: error.message }); } }); app.get('/api/:collection', async (req, res) => { try { const { collection } = req.params; const query = req.query; const result = await db.find(collection, query); res.json(result); } catch (error) { res.status(400).json({ error: error.message }); } }); app.get('/api/:collection/:id', async (req, res) => { try { const { collection, id } = req.params; const result = await db.findById(collection, id); if (!result) { return res.status(404).json({ error: 'Document not found' }); } res.json(result); } catch (error) { res.status(400).json({ error: error.message }); } }); app.put('/api/:collection/:id', async (req, res) => { try { const { collection, id } = req.params; const updated = await db.updateById(collection, id, req.body); if (!updated) { return res.status(404).json({ error: 'Document not found' }); } const result = await db.findById(collection, id); res.json(result); } catch (error) { res.status(400).json({ error: error.message }); } }); app.delete('/api/:collection/:id', async (req, res) => { try { const { collection, id } = req.params; const deleted = await db.deleteById(collection, id); if (deleted === 0) { return res.status(404).json({ error: 'Document not found' }); } res.json({ message: 'Document deleted successfully' }); } catch (error) { res.status(400).json({ error: error.message }); } }); // Specialized endpoints for common operations app.get('/api/:collection/search', async (req, res) => { try { const { collection } = req.params; const { field, value, limit = 10 } = req.query; if (!field || !value) { return res.status(400).json({ error: 'field and value are required for search' }); } const collectionData = await db.getCollection(collection); const searchValue = value.toLowerCase().trim(); const results = collectionData.filter(doc => { const fieldValue = doc[field]; if (typeof fieldValue === 'string') { return fieldValue.toLowerCase().trim().includes(searchValue); } return fieldValue == value; }).slice(0, parseInt(limit)); res.json(results); } catch (error) { res.status(400).json({ error: error.message }); } }); app.get('/api/:collection/count', async (req, res) => { try { const { collection } = req.params; const query = req.query; const count = await db.count(collection, query); res.json({ count }); } catch (error) { res.status(400).json({ error: error.message }); } }); // Specialized endpoint for versions app.get('/api/versions', async (req, res) => { try { const { software_id } = req.query; if (!software_id) { return res.status(400).json({ error: 'software_id parameter is required' }); } const versions = await db.find('versions', { software_id }); res.json(versions); } catch (error) { res.status(400).json({ error: error.message }); } }); // Initialize server async function startServer() { await ensureDataDir(); app.listen(PORT, () => { console.log(`🚀 SmallerDB Server is running on http://localhost:${PORT}`); console.log(`📁 Data directory: ${DB_PATH}`); console.log(`🔍 Health check: http://localhost:${PORT}/api/health`); console.log(''); console.log('Available collections:'); console.log('- users'); console.log('- software'); console.log('- software_files'); console.log('- downloads'); console.log('- versions'); console.log(''); console.log('API Endpoints:'); console.log(`GET /api/:collection - Get all documents`); console.log(`POST /api/:collection - Create document`); console.log(`GET /api/:collection/:id - Get document by ID`); console.log(`PUT /api/:collection/:id - Update document`); console.log(`DELETE /api/:collection/:id - Delete document`); console.log(`GET /api/:collection/search - Search documents`); console.log(`GET /api/:collection/count - Count documents`); }); } startServer().catch(console.error); module.exports = { SmallerDB };
